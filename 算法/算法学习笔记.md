 class Solution {

    public int lengthOfLongestSubstring(String s) {

        Set<Character> set=new HashSet<>();//用于去重

        int result=0;int n=s.length();

        //**外层循环扩展右边界，内层循环扩展左边界**

        for (int left = 0, right = 0 ; right < n ; right++) {

            //当前考虑的元素

            char ch=s.charAt(right);

            while (set.contains(ch)) {//如果set中已经有了该字符

            }

            result=Math.max(result,right-left+1);//记录result的max

        }

        return result;
